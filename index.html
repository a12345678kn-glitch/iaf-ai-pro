#!/usr/bin/env python3 """ضغط الفيديو قبل النشر على تيك توك (1080x1920، أقل من 287MB)""" import subprocess, sys, os, argparse, json TIKTOK_MAX_MB = 287 W, H = 1080, 1920 def get_duration(path): r = subprocess.run(["ffprobe","-v","quiet","-print_format","json", "-show_format",path], capture_output=True, text=True) return float(json.loads(r.stdout)["format"]["duration"]) def size_mb(path): return os.path.getsize(path) / (1024*1024) def compress(inp, out=None, crf=23, preset="medium"): if not os.path.isfile(inp): print(f"خطأ: الملف غير موجود {inp}"); sys.exit(1) for t in ["ffmpeg","ffprobe"]: try: subprocess.run([t,"-version"], capture_output=True, check=True) except (subprocess.CalledProcessError, FileNotFoundError): print(f"خطأ: يجب تثبيت {t}"); sys.exit(1) if out is None: base, _ = os.path.splitext(inp) out = f"{base}_tiktok.mp4" orig = size_mb(inp) dur = get_duration(inp) print(f"الأصلي: {orig:.1f} MB | المدة: {dur:.1f} ثانية") print(f"الهدف: {W}x{H} | H.264 | < {TIKTOK_MAX_MB} MB") cmd = [ "ffmpeg","-y","-i",inp, "-c:v","libx264","-crf",str(crf),"-preset",preset, "-vf", (f"scale={W}:{H}:force_original_aspect_ratio=decrease," f"pad={W}:{H}:(ow-iw)/2:(oh-ih)/2:black"), "-c:a","aac","-b:a","128k","-ar","44100", "-movflags","+faststart","-map_metadata","-1", out ] print(f"جاري الضغط إلى {out} ...") r = subprocess.run(cmd, capture_output=True, text=True) if r.returncode != 0: print(f"خطأ FFmpeg:\n{r.stderr}"); sys.exit(1) new = size_mb(out) ratio = (1 - new/orig) * 100 if orig > 0 else 0 print(f"\nتم!") print(f" الملف: {out}") print(f" الأصلي: {orig:.1f} MB") print(f" المضغوط: {new:.1f} MB") print(f" نسبة الضغط: {ratio:.1f}%") if new > TIKTOK_MAX_MB: print(f"\nتحذير: يتجاوز {TIKTOK_MAX_MB} MB") print(f"جرّب: python {sys.argv[0]} \"{inp}\" -c {crf+4}") else: print("\n✓ مطابق لمتطلبات تيك توك") def main(): p = argparse.ArgumentParser(description="ضغط فيديو لتيك توك") p.add_argument("input", help="ملف الفيديو المدخل") p.add_argument("-o","--output", help="ملف الإخراج") p.add_argument("-c","--crf", type=int, default=23, help="جودة CRF (افتراضي 23، تيك توك 22-28)") p.add_argument("-p","--preset", default="medium", choices=["ultrafast","superfast","veryfast","faster", "fast","medium","slow","slower","veryslow"], help="سرعة الترميز (افتراضي medium)") a = p.parse_args() compress(a.input, a.output, a.crf, a.preset) if __name__ == "__main__": main()
