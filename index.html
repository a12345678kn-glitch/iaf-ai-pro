const express = require('express'); const multer = require('multer'); const ffmpeg = require('fluent-ffmpeg'); const ffmpegInstaller = require('@ffmpeg-installer/ffmpeg'); const fs = require('fs-extra'); const path = require('path'); // تعيين مسار FFmpeg تلقائياً ffmpeg.setFfmpegPath(ffmpegInstaller.path); const app = express(); const PORT = 3000; // إعداد مجلدات الرفع والمعالجة const uploadDir = path.join(__dirname, 'uploads'); const outputDir = path.join(__dirname, 'processed'); fs.ensureDirSync(uploadDir); fs.ensureDirSync(outputDir); const upload = multer({ dest: uploadDir }); // واجهة HTML بسيطة للرفع app.get('/', (req, res) => { res.send(` معالجة فيديوهات تيك توك فائقة الجودة

معالج تيك توك الاحترافي (FFmpeg)

`); }); // مسار معالجة الفيديو app.post('/process', upload.single('video'), (req, res) => { if (!req.file) { return res.status(400).send('الرجاء رفع ملف فيديو صالح.'); } const inputPath = req.file.path; const outputFileName = `HQ_TikTok_${Date.now()}.mp4`; const outputPath = path.join(outputDir, outputFileName); console.log('--- بدأت عملية معالجة وضغط الفيديو ---'); ffmpeg(inputPath) .outputOptions([ // إجبار الأبعاد على 1080x1920 مع إضافة خلفية ضبابية أو سوداء للمحافظة على نسبة العرض دون تشوه '-vf scale=1080:1920:force_original_aspect_ratio=decrease,pad=1080:1920:(ow-iw)/2:(oh-ih)/2:color=black', // كوديك الفيديو القياسي والأكثر توافقاً مع تيك توك وجودته عالية '-c:v libx264', // بريزيت معتدل يوازن بين سرعة المعالجة وجودة الضغط الحادة '-preset slow', // معدل الثبات الثابت CRF 22 لضمان عدم ملاحظة أي فقدان في الجودة مع تقليص الحجم الذكي '-crf 22', // تثبيت خوارزمية الترميز لتتوافق مع كل الأجهزة (Main Profile) '-profile:v main', '-level 4.0', // كوديك الصوت عالي النقاء '-c:a aac', '-b:a 192k', '-ar 44100', // ضمان التوافقية وسرعة تشغيل الفيديو فور فتحه (Faststart) '-movflags +faststart', // تقييد استخدام الخيوط لضمان استقرار السيرفر '-threads 0' ]) .on('progress', (progress) => { if (progress.percent) { console.log(`نسبة الإنجاز: ${Math.round(progress.percent)}%`); } }) .on('error', (err, stdout, stderr) => { console.error('خطأ في معالجة FFmpeg:', err.message); console.error('تفاصيل الخطأ:', stderr); fs.removeSync(inputPath); res.status(500).send('حدث خطأ أثناء معالجة الفيديو.'); }) .on('end', () => { console.log('اكتملت المعالجة بنجاح!'); // تنظيف الملف الأصلي المرفوع fs.removeSync(inputPath); // إرسال الملف المعالج للتحميل المباشر res.download(outputPath, outputFileName, (err) => { if (err) console.error('خطأ أثناء إرسال الملف:', err); // حذف الملف النهائي بعد تنزيله لتوفير مساحة السيرفر setTimeout(() => fs.removeSync(outputPath), 10000); }); }) .save(outputPath); }); app.listen(PORT, () => { console.log(`السيرفر يعمل بكفاءة على الرابط: http://localhost:${PORT}`); });
